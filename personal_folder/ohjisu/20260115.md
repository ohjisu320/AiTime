    - 비디오: [`react-media-recorder` `video: true` ] [`MediaRecorder API` → `FormData`]
    
    [MediaRecorder - Web APIs | MDN](https://developer.mozilla.org/en-US/docs/Web/API/MediaRecorder)
    
    - 구현 코드
        
        ```tsx
        import { useRef, useState } from 'react';
        import axios from 'axios';
        
        export const VideoRecorder = () => {
          const videoRef = useRef<HTMLVideoElement>(null);
          const mediaRecorderRef = useRef<MediaRecorder | null>(null);
          const [chunks, setChunks] = useState<Blob[]>([]);
          const [isRecording, setIsRecording] = useState(false);
        
          // 1. 카메라 켜기
          const startCamera = async () => {
            try {
              const stream = await navigator.mediaDevices.getUserMedia({ video: true, audio: true });
              if (videoRef.current) {
                videoRef.current.srcObject = stream;
              }
              return stream;
            } catch (err) {
              console.error("카메라 접근 실패:", err);
            }
          };
        
          // 2. 녹화 시작
          const startRecording = async () => {
            const stream = await startCamera();
            if (!stream) return;
        
            // MediaRecorder 생성 (webm 형식으로 녹화됨)
            const recorder = new MediaRecorder(stream, { mimeType: 'video/webm' });
            mediaRecorderRef.current = recorder;
        
            // 데이터가 쌓일 때마다 배열에 저장
            recorder.ondataavailable = (event) => {
              if (event.data.size > 0) {
                setChunks((prev) => [...prev, event.data]);
              }
            };
        
            recorder.start();
            setIsRecording(true);
          };
        
          // 3. 녹화 종료 및 서버 전송
          const stopAndUpload = () => {
            const recorder = mediaRecorderRef.current;
            if (!recorder) return;
        
            recorder.onstop = async () => {
              // A. 녹화된 조각들을 하나로 합침 (Blob 생성)
              const videoBlob = new Blob(chunks, { type: 'video/webm' });
              
              // B. FormData(택배 박스)에 담기
              const formData = new FormData();
              // 'file'은 Spring Boot 컨트롤러에서 받을 변수명입니다.
              formData.append('file', videoBlob, 'baby_action.webm');
              // 필요한 경우 추가 정보도 같이 보낼 수 있음
              formData.append('userId', 'jisu123'); 
        
              // C. 서버로 전송
              try {
                await axios.post('http://localhost:8080/api/video/upload', formData, {
                  headers: {
                    'Content-Type': 'multipart/form-data', // 필수 헤더
                  },
                });
                alert("서버 전송 성공!");
              } catch (error) {
                console.error("업로드 실패:", error);
              }
              
              // 초기화
              setChunks([]);
            };
        
            recorder.stop();
            setIsRecording(false);
            
            // 카메라 끄기 (선택사항)
            const stream = videoRef.current?.srcObject as MediaStream;
            stream?.getTracks().forEach(track => track.stop());
          };
        
          return (
            <div>
              <video ref={videoRef} autoPlay playsInline muted style={{ width: '300px', background: 'black' }} />
              <br />
              {!isRecording ? (
                <button onClick={startRecording}>녹화 시작</button>
              ) : (
                <button onClick={stopAndUpload} style={{ color: 'red' }}>녹화 종료 & 전송</button>
              )}
            </div>
          );
        };
        ```
